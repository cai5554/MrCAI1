# AD5450 + AD9833 Integration Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Integrate AD5450 DAC and AD9833 DDS into one STM32H723 project, sharing SCLK/SDATA with separate CS lines.

**Architecture:** Shared SPI bus layer (`spi_bus.c/.h`) provides common bit-bang 16-bit write. AD5450 and AD9833 drivers each manage their own CS pin and call the shared layer for data transfer. Main.c initializes both and outputs fixed test values.

**Tech Stack:** STM32H723ZGT6, STM32 HAL, Keil MDK-ARM, bare-metal (no RTOS)

#define SPI_SCLK_LOW()    HAL_GPIO_WritePin(SCLK_GPIO_Port, SCLK_Pin, GPIO_PIN_RESET)
#define SPI_SDATA_HIGH()  HAL_GPIO_WritePin(SDATA_GPIO_Port, SDATA_Pin, GPIO_PIN_SET)
#define SPI_SDATA_LOW()   HAL_GPIO_WritePin(SDATA_GPIO_Port, SDATA_Pin, GPIO_PIN_RESET)

void SPI_Bus_Init(void);
void SPI_Bus_Write16(uint16_t data);

#endif /* __SPI_BUS_H */
```

- [ ] **Step 2: Create spi_bus.c**

Create file `H723_AD5450_AD9833/Hardware/spi_bus.c`:

```c
#include "spi_bus.h"

static void spi_delay(void)
{
    volatile uint32_t i = 20;
    while (i--);
}

/* Set bus to idle state: SCLK=HIGH, SDATA=LOW */
void SPI_Bus_Init(void)
{
    SPI_SCLK_HIGH();
    SPI_SDATA_LOW();
}

/*
 * Clock out 16 bits, MSB first. CPOL=1, CPHA=0.
 * Does NOT manage CS -- caller is responsible.
 *
 *   SCLK  ````````````\_____/```\_____/```\___
 *   SDIN  =======D15====X==D14===X==D0===X====
 *                      ↑       ↑       ↑
 *                    采样    采样    采样
 */
void SPI_Bus_Write16(uint16_t data)
{
    uint8_t i;

    for (i = 0; i < 16; i++) {
        /* Set data bit */
        if (data & 0x8000) {
            SPI_SDATA_HIGH();
        } else {
            SPI_SDATA_LOW();
        }
        spi_delay();

        /* Falling edge: slave samples */
        SPI_SCLK_LOW();
        spi_delay();

        /* Rising edge: prepare for next bit */
        SPI_SCLK_HIGH();
        spi_delay();

        data <<= 1;
    }
}
```

#include "spi_bus.h"

/* Initialize: ensure SPI bus idle, CS deselected */
void AD5450_Init(void)
{
    SPI_Bus_Init();
    HAL_GPIO_WritePin(AD5450_SYNC_GPIO_Port, AD5450_SYNC_Pin, GPIO_PIN_SET);
}

/* Write 8-bit DAC value (0~255) */
void AD5450_SetValue(uint8_t value)
{
    uint16_t frame = (uint16_t)value << 6;  /* [15:14]=00 control, [13:6]=data */

    HAL_GPIO_WritePin(AD5450_SYNC_GPIO_Port, AD5450_SYNC_Pin, GPIO_PIN_RESET);
    SPI_Bus_Write16(frame);
    HAL_GPIO_WritePin(AD5450_SYNC_GPIO_Port, AD5450_SYNC_Pin, GPIO_PIN_SET);
}
```

#ifndef _AD9833_SPI_H
#define _AD9833_SPI_H

#include "main.h"

/* AD9833 Register addresses */
#define AD9833_REG_CMD      (0 << 14)
#define AD9833_REG_FREQ0    (1 << 14)
#define AD9833_REG_FREQ1    (2 << 14)
#define AD9833_REG_PHASE0   (6 << 13)
#define AD9833_REG_PHASE1   (7 << 13)

/* Control bits */
#define AD9833_B28          (1 << 13)
#define AD9833_HLB          (1 << 12)
#define AD9833_FSEL0        (0 << 11)
#define AD9833_FSEL1        (1 << 11)
#define AD9833_PSEL0        (0 << 10)
#define AD9833_PSEL1        (1 << 10)
#define AD9833_PIN_SW       (1 << 9)
#define AD9833_RESET        (1 << 8)
#define AD9833_SLEEP1       (1 << 7)
#define AD9833_SLEEP12      (1 << 6)
#define AD9833_OPBITEN      (1 << 5)
#define AD9833_SIGN_PIB     (1 << 4)
#define AD9833_DIV2         (1 << 3)
#define AD9833_MODE         (1 << 1)

/* Waveform types */
#define AD9833_OUT_SINUS    ((0 << 5) | (0 << 1) | (0 << 3))
#define AD9833_OUT_TRIANGLE ((0 << 5) | (1 << 1) | (0 << 3))
#define AD9833_OUT_MSB      ((1 << 5) | (0 << 1) | (1 << 3))
#define AD9833_OUT_MSB2     ((1 << 5) | (0 << 1) | (0 << 3))

/* Function declarations */
void AD9833_Init(void);
void AD9833_Reset(void);
void AD9833_ClearReset(void);
void AD9833_SetRegisterValue(uint16_t regValue);
void AD9833_SetFrequency(uint16_t reg, uint16_t wave_type, uint32_t val);
void AD9833_SetPhase(uint16_t reg, uint16_t val);
void AD9833_Setup(uint16_t freq, uint16_t phase, uint16_t type);
void AD9833_SetWave(uint16_t type);

#endif /* _AD9833_SPI_H */
```

- [ ] **Step 2: Create AD9833_SPI.c**

Create file `H723_AD5450_AD9833/Hardware/AD9833_SPI.c`:

```c
#include "AD9833_SPI.h"
#include "spi_bus.h"

/* Initialize: reset the DDS */
void AD9833_Init(void)
{
    SPI_Bus_Init();
    AD9833_SetRegisterValue(AD9833_REG_CMD | AD9833_RESET);
}

/* Reset DDS output to midscale */
void AD9833_Reset(void)
{
    AD9833_SetRegisterValue(AD9833_REG_CMD | AD9833_RESET);
}

/* Clear reset, start output */
void AD9833_ClearReset(void)
{
    AD9833_SetRegisterValue(AD9833_REG_CMD);
}

/* Write 16-bit value to AD9833 via SPI with CS control */
void AD9833_SetRegisterValue(uint16_t regValue)
{
    HAL_GPIO_WritePin(AD9833_SYNC_GPIO_Port, AD9833_SYNC_Pin, GPIO_PIN_RESET);
    SPI_Bus_Write16(regValue);
    HAL_GPIO_WritePin(AD9833_SYNC_GPIO_Port, AD9833_SYNC_Pin, GPIO_PIN_SET);
}

/* Set frequency register (28-bit value, written in two 14-bit chunks via B28) */
void AD9833_SetFrequency(uint16_t reg, uint16_t wave_type, uint32_t val)
{
    uint16_t freqHi = reg;
    uint16_t freqLo = reg;

    freqHi |= (val & 0xFFFC000) >> 14;
    freqLo |= (val & 0x3FFF);

    AD9833_SetRegisterValue(AD9833_B28);
    AD9833_SetRegisterValue(freqLo);
    AD9833_SetRegisterValue(freqHi);
    AD9833_SetRegisterValue(AD9833_REG_CMD | AD9833_B28 | wave_type);
}

/* Set phase register (12-bit value) */
void AD9833_SetPhase(uint16_t reg, uint16_t val)
{
    uint16_t phase = reg;
    phase |= val;
    AD9833_SetRegisterValue(phase);
}

/* Set frequency register, phase register, and waveform type */
void AD9833_Setup(uint16_t freq, uint16_t phase, uint16_t type)
{
    uint16_t val = freq | phase | type;
    AD9833_SetRegisterValue(val);
}

/* Set waveform type directly */
void AD9833_SetWave(uint16_t type)
{
    AD9833_SetRegisterValue(type);
}
```
